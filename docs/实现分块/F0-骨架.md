### 4.1 新建 `src/api.js`

```javascript
/**
 * 【功能】把后端 13 个接口封装成 13 个函数，页面里只写 api.xxx()
 * 【作用】统一 baseURL、超时、错误提示，避免每个页面各写一套 axios 配置
 * 【输入】各函数的参数（文件、taskId、过滤条件等）
 * 【输出】Promise，resolve 的是后端返回的 JSON（已经是对象，不是 axios 包装）
 * 【怎么用】import api from '@/api' 然后 await api.ping()
 * 【例子】const r = await api.upload(file, p => (percent.value = p))
 *        → r.taskId 就是任务号
 */
import axios from 'axios'

const http = axios.create({
  // 开发期由 vite.config.js 把 /api 代理到 http://localhost:8080
  baseURL: '/api',
  // 上传接口内部会同步跑完审计，10~60 秒很常见，短了必然假超时
  timeout: 120000,
})

http.interceptors.response.use(
  (resp) => resp.data,
  (err) => {
    const msg =
      err.response?.data?.error ||
      (err.code === 'ECONNABORTED'
        ? '请求超时：审计可能还在跑，或文件太大，请稍后刷新任务列表'
        : err.message || '请求失败')
    return Promise.reject(new Error(msg))
  },
)

export default {
  /** 探活：返回 {status, db, time} */
  ping: () => http.get('/ping'),

  /** 上传并审计（同步），onProgress 收到 0~100 */
  upload: (file, onProgress) => {
    const form = new FormData()
    form.append('file', file)
    return http.post('/upload', form, {
      headers: { 'Content-Type': 'multipart/form-data' },
      onUploadProgress: (e) => {
        if (onProgress && e.total) onProgress(Math.round((e.loaded / e.total) * 100))
      },
    })
  },

  /** 任务列表 */
  tasks: () => http.get('/audit/tasks'),
  /** 单任务：{task, violations, checked} */
  task: (id) => http.get(`/audit/tasks/${id}`),
  /** 违规明细（按 startLine 升序） */
  findings: (id) => http.get(`/audit/tasks/${id}/findings`),

  /** 报告 JSON（含 markdown / html 字符串） */
  report: (id) => http.get(`/report/${id}`),
  /** 报告 Markdown 纯文本 */
  reportMarkdown: (id) => http.get(`/report/${id}/markdown`),
  /** 报告 HTML 纯文本 */
  reportHtml: (id) => http.get(`/report/${id}/html`),
  /** 强制重算报告 */
  regenerate: (id) => http.post(`/report/${id}/regenerate`),

  /** 规则列表 */
  rules: (language, limit = 20) => http.get('/rules', { params: { language, limit } }),
  /** 规则库统计 */
  ruleStats: () => http.get('/rules/stats'),
  /** 从 JSON 文件导入规则（管理用） */
  importRules: (path) => http.post('/rules/import', null, { params: { path } }),
  /** 从规范文档蒸馏规则 */
  distill: (docPath, language, startChunk = 0, maxChunks = 5) =>
    http.post('/rules/distill', null, { params: { docPath, language, startChunk, maxChunks } }),
}
```

### 4.2 新建 `src/router.js`

```javascript
/**
 * 【功能】定义 4 条路由：上传页、任务列表、报告页、规则库
 * 【作用】地址栏变了就换组件，不用刷新页面
 * 【输入】无
 * 【输出】router 实例，在 main.js 里 app.use(router)
 * 【怎么用】页面里 const router = useRouter(); router.push('/report/6')
 * 【例子】<router-link to="/tasks">任务列表</router-link>
 */
import { createRouter, createWebHashHistory } from 'vue-router'

// 用 hash 模式（地址带 #），这样打包塞进 Spring Boot static 后，
// 刷新页面不会因为后端没有对应路由而 404。
const router = createRouter({
  history: createWebHashHistory(),
  routes: [
    { path: '/', name: 'upload', component: () => import('@/views/UploadView.vue') },
    { path: '/tasks', name: 'tasks', component: () => import('@/views/TaskListView.vue') },
    {
      path: '/report/:id',
      name: 'report',
      component: () => import('@/views/ReportView.vue'),
      props: true, // 让组件通过 props 拿到 id
    },
    { path: '/rules', name: 'rules', component: () => import('@/views/RulesView.vue') },
    { path: '/:pathMatch(.*)*', redirect: '/' },
  ],
})

export default router
```

### 4.3 覆盖 `src/App.vue`

```vue
<!--
【功能】全站外框：左侧菜单 + 顶栏（标题和后端状态灯）+ 右侧路由出口
【作用】所有页面共用，避免每页重复写导航
【输入】无
【输出】渲染当前路由对应的页面
【怎么用】不用手动引，main.js 挂载 App，路由自动填充 <router-view />
【例子】点"任务列表"→ 地址变 /#/tasks → 右侧显示 TaskListView
-->
<template>
  <el-container class="layout">
    <el-aside width="200px" class="aside">
      <div class="brand">智能代码审计</div>
      <el-menu :default-active="activeMenu" router class="menu">
        <el-menu-item index="/">上传审计</el-menu-item>
        <el-menu-item index="/tasks">任务列表</el-menu-item>
        <el-menu-item index="/rules">规则库</el-menu-item>
      </el-menu>
    </el-aside>

    <el-container>
      <el-header class="header">
        <span class="title">基于大语言模型的智能代码审计系统</span>
        <span class="status">
          <i class="dot" :class="online ? 'ok' : 'bad'" />
          {{ online ? '后端已连接' : '后端未连接' }}
        </span>
      </el-header>
      <el-main class="main">
        <router-view />
      </el-main>
    </el-container>
  </el-container>
</template>

<script setup>
import { onBeforeUnmount, onMounted, ref } from 'vue'
import { useRoute } from 'vue-router'
import api from '@/api'

const route = useRoute()
const online = ref(false)
let timer = null

// 报告页不属于菜单项，让它高亮"任务列表"
const activeMenu = ref('/')

// 每 20 秒探一次活，答辩演示时能直观看到后端是否还活着
async function checkHealth() {
  try {
    const r = await api.ping()
    online.value = r.db === 'connected'
  } catch {
    online.value = false
  }
}

onMounted(() => {
  activeMenu.value = route.path.startsWith('/report') ? '/tasks' : route.path
  checkHealth()
  timer = setInterval(checkHealth, 20000)
})

onBeforeUnmount(() => timer && clearInterval(timer))
</script>

<style scoped>
.layout { height: 100vh; }
.aside { background: #1f2d3d; color: #fff; }
.brand { height: 56px; line-height: 56px; text-align: center; font-size: 16px; color: #fff; letter-spacing: 1px; }
.menu { border-right: none; background: transparent; }
.menu :deep(.el-menu-item) { color: #c0c4cc; }
.menu :deep(.el-menu-item.is-active) { color: #fff; background: #263445; }
.header { display: flex; align-items: center; justify-content: space-between; background: #fff; border-bottom: 1px solid #ebeef5; }
.title { font-size: 15px; color: #303133; }
.status { font-size: 13px; color: #606266; }
.dot { display: inline-block; width: 8px; height: 8px; border-radius: 50%; margin-right: 6px; }
.dot.ok { background: #67c23a; }
.dot.bad { background: #f56c6c; }
.main { background: #f5f7fa; padding: 20px; overflow-y: auto; }
</style>
```

> 注意：`activeMenu` 只在 `onMounted` 算了一次，切页面时菜单高亮不会跟着变。这没关系——`el-menu` 的 `default-active` 只在首次生效，实际高亮由 Element Plus 根据点击维护。若发现刷新后高亮不对，把 `:default-active="activeMenu"` 换成 `:default-active="$route.path"` 即可。

## 6. 卡住怎么办

| 症状 | 原因 | 处理 |
|---|---|---|
| 页面全白，控制台报 `Failed to resolve component: el-container` | 没挂 Element Plus | 检查 `main.js` 里有没有 `.use(ElementPlus)`；改动后要重启 `npm run dev` |
| 报 `Cannot find module '@/api'` | 没配 `@` 别名 | 检查 `vite.config.js` 里 `resolve.alias` 那一段 |
| 报 `The requested module '/src/api.js' does not provide an export named 'default'` | api.js 写成了具名导出 | 文件末尾必须是 `export default {...}` |
| 状态灯一直红，但浏览器直接访问 8080/api/ping 是通的 | 代理没生效 | 重启 `npm run dev`；确认 `vite.config.js` 的 `server.proxy` 写的是 `/api` |
| 左侧菜单点不动 | `el-menu` 少了 `router` 属性 | 模板里必须是 `<el-menu ... router>` |
| 样式全乱、内容挤在中间 1280px | `style.css` 还是模板自带的 | 按 4.5 覆盖 |
