### 4.1 新建 `src/components/CodeBlock.vue`

```vue
<!--
【功能】渲染一段代码，带行号和语法高亮
【作用】报告页里把"第 N 行"和实际代码视觉对齐，评委一眼能看懂
【输入】code（代码文本）、language（java/python）、startLine（首行行号，1-based）
【输出】左侧行号 + 右侧高亮代码
【怎么用】<CodeBlock :code="f.codeSnippet" :language="task.language" :start-line="f.startLine" />
【例子】startLine=11 时，第一行左侧显示 11
-->
<template>
  <div class="code-block">
    <pre><code v-html="html" /></pre>
  </div>
</template>

<script setup>
import { computed } from 'vue'
import hljs from 'highlight.js/lib/core'
import java from 'highlight.js/lib/languages/java'
import python from 'highlight.js/lib/languages/python'
import 'highlight.js/styles/github.css'

hljs.registerLanguage('java', java)
hljs.registerLanguage('python', python)

const props = defineProps({
  code: { type: String, default: '' },
  language: { type: String, default: 'java' },
  // 只用于展示：把原始行号标在左侧（见下方 lines 计算）
  startLine: { type: Number, default: 1 },
})

// 每一行前面补上行号，再用 highlight.js 整体染色
const html = computed(() => {
  const raw = props.code || ''
  const lang = props.language === 'python' ? 'python' : 'java'
  let highlighted
  try {
    highlighted = hljs.highlight(raw, { language: lang }).value
  } catch {
    highlighted = raw.replace(/[<>&]/g, (c) => ({ '<': '&lt;', '>': '&gt;', '&': '&amp;' }[c]))
  }
  const hlLines = highlighted.split('\n')
  const rawLines = raw.split('\n')
  const width = String(props.startLine + rawLines.length - 1).length
  return hlLines
    .map((line, i) => {
      const no = String(props.startLine + i).padStart(width, ' ')
      return `<span class="ln">${no}</span>${line || ' '}`
    })
    .join('\n')
})
</script>

<style scoped>
.code-block {
  background: #fafafa;
  border: 1px solid #ebeef5;
  border-radius: 6px;
  overflow: auto;
  max-height: 420px;
}
.code-block pre { margin: 0; padding: 12px 14px; }
.code-block code { font-family: Consolas, "Courier New", monospace; font-size: 13px; line-height: 1.7; }
.code-block :deep(.ln) {
  display: inline-block;
  width: 2.6em;
  margin-right: 10px;
  color: #b1b3b8;
  text-align: right;
  user-select: none;
  border-right: 1px solid #e4e7ed;
  padding-right: 8px;
}
</style>
```

### 4.2 覆盖 `src/views/ReportView.vue`

```vue
<!--
【功能】展示单个任务的审计报告：合规率、结论、图表、逐条违规（行号+代码+原因+修法）
【作用】系统结论的最终呈现页，也是答辩演示重点
【输入】路由参数 id（任务号）
【输出】页面；可下载 Markdown、新窗口打开 HTML、重新生成报告
【怎么用】从任务列表点击进入，或直接访问 /#/report/6
【例子】打开 /#/report/6 → 98.7% 合规率 → 3 条明细，其中第 11 行 SQL 注入为 HIGH
-->
<template>
  <div class="page" v-loading="loading">
    <!-- 顶部：任务信息 + 三个动作 -->
    <el-card shadow="never" class="head">
      <div class="head-row">
        <div>
          <span class="file">{{ task?.fileName || '—' }}</span>
          <el-tag v-if="task" class="ml" :type="task.language === 'java' ? 'warning' : 'success'">
            {{ task.language }}
          </el-tag>
          <span class="meta">完成于 {{ fmt(task?.finishedAt) }} · 任务 #{{ id }}</span>
        </div>
        <div class="acts">
          <el-button @click="downloadMarkdown">下载 Markdown</el-button>
          <el-button @click="openHtml">投屏 HTML</el-button>
          <el-button :loading="regenerating" @click="regenerate">重新生成</el-button>
        </div>
      </div>
    </el-card>

    <!-- 合规率 + 结论 -->
    <el-row :gutter="16" class="row">
      <el-col :span="6">
        <el-card shadow="never" class="rate-card">
          <el-progress
            type="circle"
            :percentage="ratePercent"
            :width="120"
            :color="rateColor"
          />
          <div class="rate-label">合规率</div>
          <div class="rate-sub">{{ findings.length }} 条违规 / {{ ruleTotal }} 条规则</div>
        </el-card>
      </el-col>
      <el-col :span="18">
        <el-card shadow="never" class="summary-card">
          <div class="summary-title">总体评估</div>
          <p class="summary-text">{{ report?.summaryText || '暂无结论' }}</p>
          <div class="badges">
            <el-tag type="danger">HIGH {{ severityCount.HIGH || 0 }}</el-tag>
            <el-tag type="warning" class="ml">MEDIUM {{ severityCount.MEDIUM || 0 }}</el-tag>
            <el-tag type="info" class="ml">LOW {{ severityCount.LOW || 0 }}</el-tag>
          </div>
        </el-card>
      </el-col>
    </el-row>

    <!-- 两个图表 -->
    <el-row :gutter="16" class="row">
      <el-col :span="12">
        <el-card shadow="never">
          <template #header>严重程度分布</template>
          <div ref="pieRef" class="chart"></div>
        </el-card>
      </el-col>
      <el-col :span="12">
        <el-card shadow="never">
          <template #header>各规则命中条数</template>
          <div ref="barRef" class="chart"></div>
        </el-card>
      </el-col>
    </el-row>

    <!-- 明细 -->
    <el-card shadow="never" class="row">
      <template #header>
        违规明细（{{ findings.length }}）
      </template>

      <el-empty v-if="!findings.length" description="没有发现问题：这个文件在这套规则下是干净的" />

      <div v-for="f in findings" :key="f.id" class="finding">
        <div class="f-head">
          <el-tag :type="sevType(f.severity)" effect="dark">{{ f.severity }}</el-tag>
          <span class="f-line">第 {{ lineText(f) }} 行</span>
          <el-tag v-if="f.confidence < 0.5" type="warning" effect="plain" class="ml">待人工确认</el-tag>
          <span class="f-conf">置信度 {{ Math.round((f.confidence || 0) * 100) }}%</span>
          <span class="f-rule">{{ f.ruleId }}</span>
        </div>

        <CodeBlock :code="f.codeSnippet" :language="task?.language || 'java'" :start-line="f.startLine" />

        <div class="f-body">
          <div class="f-item">
            <span class="label">为什么判违规</span>
            <span class="value">{{ f.reason }}</span>
          </div>
          <div class="f-item">
            <span class="label">怎么修</span>
            <span class="value">{{ f.fixSuggestion }}</span>
          </div>
        </div>
      </div>
    </el-card>
  </div>
</template>

<script setup>
import { computed, nextTick, onBeforeUnmount, onMounted, ref } from 'vue'
import { useRoute } from 'vue-router'
import { ElMessage } from 'element-plus'
import * as echarts from 'echarts'
import api from '@/api'
import CodeBlock from '@/components/CodeBlock.vue'

const route = useRoute()
const id = route.params.id

const loading = ref(false)
const regenerating = ref(false)
const task = ref(null)
const findings = ref([])
const report = ref(null)
const stats = ref(null) // 规则库统计，用来算"x/总规则数"

const pieRef = ref(null)
const barRef = ref(null)
let pie = null
let bar = null

const sevType = (s) => ({ HIGH: 'danger', MEDIUM: 'warning', LOW: 'info' }[s] || 'info')

const lineText = (f) => (f.startLine === f.endLine ? f.startLine : `${f.startLine}–${f.endLine}`)

const ratePercent = computed(() => Math.round((report.value?.complianceRate ?? 0) * 1000) / 10)

const rateColor = computed(() => {
  const p = ratePercent.value
  if (p >= 95) return '#67c23a'
  if (p >= 80) return '#e6a23c'
  return '#f56c6c'
})

const ruleTotal = computed(() => {
  const lang = task.value?.language
  if (!lang || !stats.value?.byLanguage) return '—'
  return stats.value.byLanguage[lang] ?? '—'
})

const severityCount = computed(() => {
  const m = {}
  for (const f of findings.value) m[f.severity] = (m[f.severity] || 0) + 1
  return m
})

function fmt(v) {
  if (!v) return '—'
  const d = new Date(v)
  const p = (n) => String(n).padStart(2, '0')
  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())} ${p(d.getHours())}:${p(d.getMinutes())}`
}

async function load() {
  loading.value = true
  try {
    const [detail, fs, rep, st] = await Promise.all([
      api.task(id),
      api.findings(id),
      api.report(id),
      api.ruleStats().catch(() => null),
    ])
    task.value = detail.task
    findings.value = fs
    report.value = rep
    stats.value = st
    await nextTick()
    drawPie()
    drawBar()
  } catch (e) {
    ElMessage.error(e.message)
  } finally {
    loading.value = false
  }
}

function drawPie() {
  if (!pieRef.value) return
  pie = pie || echarts.init(pieRef.value)
  const data = ['HIGH', 'MEDIUM', 'LOW', 'INFO']
    .map((k) => ({ name: k, value: severityCount.value[k] || 0 }))
    .filter((d) => d.value > 0)
  pie.setOption({
    tooltip: { trigger: 'item' },
    legend: { bottom: 0 },
    color: ['#f56c6c', '#e6a23c', '#909399', '#409eff'],
    series: [{ type: 'pie', radius: ['40%', '65%'], data, label: { formatter: '{b} {c}' } }],
  }, true)
}

function drawBar() {
  if (!barRef.value) return
  bar = bar || echarts.init(barRef.value)
  const m = {}
  for (const f of findings.value) m[f.ruleId] = (m[f.ruleId] || 0) + 1
  const entries = Object.entries(m).sort((a, b) => b[1] - a[1]).slice(0, 8)
  bar.setOption({
    grid: { left: 8, right: 16, top: 10, bottom: 8, containLabel: true },
    tooltip: { trigger: 'axis' },
    xAxis: { type: 'value', minInterval: 1 },
    yAxis: {
      type: 'category',
      data: entries.map(([k]) => (k.length > 28 ? k.slice(0, 28) + '…' : k)).reverse(),
      axisLabel: { fontSize: 11 },
    },
    series: [{ type: 'bar', data: entries.map(([, v]) => v).reverse(), barWidth: 14, itemStyle: { color: '#409eff' } }],
  }, true)
}

function resize() {
  pie?.resize()
  bar?.resize()
}

async function regenerate() {
  regenerating.value = true
  try {
    await api.regenerate(id)
    ElMessage.success('报告已重新生成')
    report.value = await api.report(id)
    await nextTick()
    drawPie()
    drawBar()
  } catch (e) {
    ElMessage.error(e.message)
  } finally {
    regenerating.value = false
  }
}

// 直接打开后端导出的资源：Markdown 会触发下载，HTML 在新窗口渲染
function downloadMarkdown() {
  window.open(`/api/report/${id}/markdown`, '_blank')
}
function openHtml() {
  window.open(`/api/report/${id}/html`, '_blank')
}

onMounted(() => {
  load()
  window.addEventListener('resize', resize)
})

onBeforeUnmount(() => {
  window.removeEventListener('resize', resize)
  pie?.dispose()
  bar?.dispose()
})
</script>

</style>
```

## 6. 卡住怎么办

| 症状 | 原因 | 处理 |
|---|---|---|
| 图表区域全白 | 容器高度为 0，或 `echarts.init` 执行太早 | 确认 `.chart{height:240px}`；确保 `await nextTick()` 之后才 init |
| 合规率显示 0% | `report.complianceRate` 字段名拼错 | 后端返回的是 `complianceRate`（驼峰） |
| 行号全从 1 开始 | 没把 `f.startLine` 传给 `CodeBlock` | 检查 `:start-line="f.startLine"` |
| 点击下载没反应 | 浏览器拦截了弹窗 | 允许弹出窗口；或改成 `window.location.href = ...` |
| 报 400 任务不存在 | 路由参数 id 不对 | 从任务列表点进来，别手敲 URL |
| `highlight.js/styles/github.css` 报找不到 | 版本差异 | 换成 `highlight.js/styles/default.css` |
| 代码块行号与代码错行 | `codeSnippet` 是多行且首行为空 | 后端已去首尾空白；若有问题可在 `html` 计算里 `raw.trimStart()` |
